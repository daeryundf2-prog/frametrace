    17	pub(super) fn resolve_validation_target(
    18	    case_dir: &Path,
    19	    selector: &str,
    20	) -> Result<ValidationTarget, String> {
    21	    let direct = PathBuf::from(selector);
    22	    if direct.is_file() {
    23	        let path = direct
    24	            .canonicalize()
    25	            .map_err(|err| format!("failed to canonicalize validation target: {err}"))?;
    26	        return Ok(ValidationTarget {
    27	            path,
    28	            source_artifact_id: None,
    29	            source_artifact_path: None,
    30	            source_artifact_sha256: None,
    31	            derived_artifact_id: None,
    32	            target_artifact_id: None,
    33	        });
    34	    }
    35	
    36	    if let Ok(path) = resolve_video_source(case_dir, selector) {
    37	        let path = path
    38	            .canonicalize()
    39	            .map_err(|err| format!("failed to canonicalize validation target: {err}"))?;
    40	        let source_artifact_sha256 = audit::indexed_source_hash(case_dir, selector, &path);
    41	        let source_artifact_id = source_artifact_sha256
    42	            .as_deref()
    43	            .map(|sha256| media_contract::source_artifact_id(selector, sha256));
    44	        return Ok(ValidationTarget {
    45	            source_artifact_path: Some(path.clone()),
    46	            path,
    47	            source_artifact_id: source_artifact_id.clone(),
    48	            source_artifact_sha256,
    49	            derived_artifact_id: None,
    50	            target_artifact_id: source_artifact_id,
    51	        });
    52	    }
    53	
    54	    for rel_log in [
    55	        "artifacts/carved/carve-log.jsonl",
    56	        "artifacts/clips/export-log.jsonl",
    57	        "artifacts/proxies/proxy-log.jsonl",
    58	        "artifacts/thumbnails/thumbnail-log.jsonl",
    59	        "artifacts/frames/frame-log.jsonl",
    60	        "evidence/logs/tsk-audit.jsonl",
    61	    ] {
    62	        let Some(mut target) = resolve_from_log(&case_dir.join(rel_log), selector) else {
    63	            continue;
    64	        };
    65	        if target.path.is_file() {
    66	            target.path = target
    67	                .path
    68	                .canonicalize()
    69	                .map_err(|err| format!("failed to canonicalize validation target: {err}"))?;
    70	            return Ok(target);
    71	        }
    72	    }
    73	
    74	    Err(format!(
    75	        "validation target not found: {selector} (use an indexed video id, artifact id, inode recovery path, or direct file path)"
    76	    ))
    77	}
    78	
    79	pub(super) fn resolve_from_log(log_path: &Path, selector: &str) -> Option<ValidationTarget> {
    80	    let text = read_to_string(log_path).ok()?;
    81	    for line in text.lines().filter(|line| !line.trim().is_empty()) {
    82	        let id = extract_json_string(line, "id");
    83	        let inode = extract_json_string(line, "inode");
    84	        let output_path = extract_json_string(line, "output_path");
    85	        let output_artifact_path = extract_json_string(line, "output_artifact_path");
    86	        let selector_field = extract_json_string(line, "selector");
    87	        let derived_artifact_id = extract_json_string(line, "derived_artifact_id");
    88	        let source_artifact_id = extract_json_string(line, "source_artifact_id");
    89	        let source_artifact_path = extract_json_string(line, "source_artifact_path")
    90	            .or_else(|| extract_json_string(line, "source_path"));
    91	        let source_artifact_sha256 = extract_json_string(line, "source_artifact_sha256")
    92	            .or_else(|| extract_json_string(line, "source_index_sha256"));
    93	        let matches = id.as_deref() == Some(selector)
    94	            || inode.as_deref() == Some(selector)
    95	            || selector_field.as_deref() == Some(selector)
    96	            || derived_artifact_id.as_deref() == Some(selector)
    97	            || source_artifact_id.as_deref() == Some(selector)
    98	            || output_path.as_deref() == Some(selector)
    99	            || output_artifact_path.as_deref() == Some(selector);
   100	        if matches {
   101	            let target_artifact_id = derived_artifact_id
   102	                .clone()
   103	                .or_else(|| id.clone())
   104	                .or_else(|| inode.clone())
   105	                .or_else(|| selector_field.clone());
   106	            return output_path
   107	                .or(output_artifact_path)
   108	                .map(|path| ValidationTarget {
   109	                    path: PathBuf::from(path),
   110	                    source_artifact_id,
   111	                    source_artifact_path: source_artifact_path.map(PathBuf::from),
   112	                    source_artifact_sha256,
   113	                    derived_artifact_id,
   114	                    target_artifact_id,
   115	                });
   116	        }
   117	    }
   118	    None
   119	}
